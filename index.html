function sendEmail(data) { MailApp.sendEmail({ to: "nurychevkaisar@gmail.com", subject: "Новая запись в Sanare", body: `Новая запись:\nВрач: ${data.doctor}\nИмя: ${data.name}\nТелефон: ${data.phone}\nДата: ${data.appointment_date}\nВремя: ${data.appointment_time}` }); } function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Sheet1'); var data = { doctor: e.parameter.doctor, name: e.parameter.name, phone: e.parameter.phone, appointment_date: e.parameter.appointment_date, appointment_time: e.parameter.appointment_time }; sheet.appendRow([data.doctor, data.name, data.phone, data.appointment_date, data.appointment_time]); sendEmail(data); return ContentService.createTextOutput('Success').setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput('Error: ' + error).setMimeType(ContentService.MimeType.TEXT); } }
